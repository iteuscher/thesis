% Shared layout: rotated figure + caption kept together on one portrait page
\newcommand{\printrotatedfigurelayout}[4]{%
    \par\vspace{0.15em}%
    \vspace*{\fill}
    \noindent\begin{minipage}{\textwidth}
        \centering
        \rotatebox{90}{%
            \includegraphics[%
                width=\printrotatedfigurewidth,%
                height=\printrotatedfigureheight,%
                keepaspectratio]{#1}%
        }%
        \par\vspace{0.25em}%
        \captionof{figure}[#2]{#3}%
        \label{#4}%
    \end{minipage}%
    \vspace*{\fill}
}

% Usage: \landscapefigurepage[image height]{image}{LOF caption}{figure caption}{label}
\newcommand{\landscapefigurepage}[5][0.78\textheight]{%
    \printrotatedfigurelayout{#2}{#3}{#4}{#5}%
}

% Appendix diagram: section title + rotated figure on one portrait page
% Usage: \landscapeappendixdiagram[image height]{section title}{section label}{image}{LOF caption}{figure caption}{figure label}
\newcommand{\landscapeappendixdiagram}[7][0.74\textheight]{%
    \section{#2}%
    \label{#3}%
    \printrotatedfigurelayout{#4}{#5}{#6}{#7}%
}

% Full-page portrait figure with lead-in text (figure title block) + image + caption
% Usage: \centeredfigurepage[width]{lead-in}{image}{LOF caption}{figure caption}{label}
\newcommand{\centeredfigurepage}[6][0.86\textwidth]{%
    \clearpage
    \begin{figure}[H]
        \centering
        \vspace*{\fill}
        \begin{minipage}{0.9\textwidth}
            \centering
            #2
        \end{minipage}
        \par\vspace{1em}
        \includegraphics[width=#1,keepaspectratio]{#3}
        \caption[#4]{#5}
        \label{#6}
        \vspace*{\fill}
    \end{figure}
}
